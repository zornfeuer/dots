### disk free space
df -h

### memory top
ps aux --sort=-%mem | head

### cpu top
ps aux --sort=-%cpu | head

### docker ps
sudo docker ps

### docker logs
sudo docker logs <container_id> --tail 200

### list ports
ss -tulpn

### delete postgres logs
sudo find /opt/edr/postgres/logs -name "*.log*" -type f -mtime +7 -delete
